// ============================================ // ไฟล์: Code.gs // ไฟล์หลักสำหรับรับ Webhook และจัดการ Events // ============================================ function doPost(e) { try { if (typeof refreshConfig_ === 'function') refreshConfig_(); const params = (e && e.parameter) || {}; const connection = getConnection_(params.bot); if (!connection) { return ContentService.createTextOutput(JSON.stringify({ status: 'BOT_NOT_FOUND' })) .setMimeType(ContentService.MimeType.JSON); } const events = JSON.parse(e.postData.contents).events || []; events.forEach(event => { try { switch (event.type) { case 'follow': handleFollowEvent(event, connection); break; case 'message': if (event.message && event.message.type === 'text') { handleMessageEvent(event, connection); } break; case 'postback': handlePostbackEvent(event, connection); break; default: Logger.log(`Unhandled event: ${event.type}`); } } catch (eventError) { Logger.log(`Error handling ${event.type} event: ${eventError}`); } }); return ContentService.createTextOutput(JSON.stringify({ status: 'OK' })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { Logger.log('Error in doPost: ' + error); return ContentService.createTextOutput(JSON.stringify({ error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function buildPromptMessageText(conn) { try { const groupSummary = getProductGroupSummary(conn); const groupDetails = Object.entries(groupSummary.groups) .map(([name, cnt]) => `• ${name} ${cnt} รายการ`) .join('\n'); let text = 'สวัสดีค่ะ ต้องการสอบถามสินค้า ราคา หรือโปรโมชั่นตัวไหน แจ้งรหัสสินค้าหรือชื่อสินค้าได้เลยค่ะ\n\n'; text += `ตอนนี้มีสินค้าหลัก ${groupSummary.count} กลุ่มค่ะ\n${groupDetails}`; return text; } catch (e) { return 'สวัสดีค่ะ ต้องการสอบถามสินค้า ราคา หรือโปรโมชั่นตัวไหน แจ้งรหัสสินค้าหรือชื่อสินค้าได้เลยค่ะ'; } } function buildPromptMessage_(conn) { const message = { type: 'text', text: buildPromptMessageText(conn) }; const quickReply = isLiffMode_() ? buildLiffQuickReply_(conn) : buildGroupQuickReply(conn); if (quickReply) message.quickReply = quickReply; return message; } // ============================================ // จัดการ Event: Follow (เพิ่มเพื่อน) // ============================================ function handleFollowEvent(event, conn) { try { const userId = event.source.userId; if (typeof registerFirstAdminUid_ === 'function') registerFirstAdminUid_(userId); // ดึงข้อมูลโปรไฟล์จาก LINE const profile = getUserProfile(conn, userId); // เตรียมข้อมูลผู้ใช้ const userData = { User_Id: userId, DisplayName: profile.displayName || 'คุณลูกค้า', ProfileUrl: profile.pictureUrl || '', Status: 'Active', ImagePicture: profile.pictureUrl || '', '#REF!': '', Date: Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'dd/MM/yyyy'), Time: Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'HH:mm:ss'), Language: profile.language || 'th' }; // บันทึกข้อมูลลง Google Sheets saveUserData(userData, conn); // Webhook records the follow event only. All chat messages are sent by // the LIFF client with liff.sendMessages(). Open LIFF from a Rich Menu or link. Logger.log(`New user followed (no webhook message sent): ${userId}`); } catch (error) { Logger.log('Error in handleFollowEvent: ' + error); } } // ============================================ // จัดการ Event: Postback (กดปุ่ม) // ============================================ function handlePostbackEvent(event, conn) { const data = event && event.postback ? event.postback.data : ''; Logger.log(`Postback received without webhook reply: ${data}`); } // ============================================ // จัดการ Event: Message (ข้อความ) // ============================================ function handleMessageEvent(event, conn) { try { const rawText = String(event.message.text || '').trim(); // ข้อความที่ LIFF ส่งเข้าห้องแชทจะมี zero-width space นำหน้า // ไม่ตอบจาก webhook เพื่อไม่ใช้ Messaging API ส่งข้อความ if (isLiffText_(rawText) || rawText === LIFF_CLOSE_MARKER) return; Logger.log(`Incoming text ignored by webhook; use LIFF to respond (${rawText.length} chars)`); } catch (error) { Logger.log('Error in handleMessageEvent: ' + error); } } // ============================================ // LINE API Functions // ============================================ function getUserProfile(conn, userId) { const url = `${CONFIG.LINE_API_URL}/profile/${userId}`; const options = { method: 'get', headers: { 'Authorization': `Bearer ${getConnectionToken_(conn)}`, 'Content-Type': 'application/json' }, muteHttpExceptions: true }; const response = UrlFetchApp.fetch(url, options); return JSON.parse(response.getContentText()); }
